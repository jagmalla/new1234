```php
// Detection
$h = houseFromMoon($shani_rashi, $natal_moon_rashi);   // existing Layer-1 helper reuse
if (in_array($h,[12,1,2]))  $status = ['type'=>'SadeSati','phase'=>[12=>1,1=>2,2=>3][$h]];
elseif (in_array($h,[4,8])) $status = ['type'=>'Dhaiyya'];

$PAYA_METAL = [1=>'Gold',6=>'Gold',11=>'Gold', 2=>'Silver',5=>'Silver',9=>'Silver',
               3=>'Copper',7=>'Copper',10=>'Copper', 4=>'Iron',8=>'Iron',12=>'Iron'];
$PAYA_GOOD  = ['Silver'=>true,'Copper'=>true,'Gold'=>false,'Iron'=>false];

// Severity (0=none .. 3=critical) — सब checks जन्म कुण्डली + current dasha से
function sadesatiSeverity($chart, $shaniTransit) {
  $s = 2; // default moderate
  if (natalShaniStrong($chart))          $s -= 2;  // uchcha/swagrihi/mitragrihi/yogakaraka
                                                    // (लग्न और चन्द्र लग्न दोनों से check)
  if (natalShaniWeakEnemySign($chart))   $s += 1;
  $rel = nakshatraLordRelationToShani($shaniTransit->nakshatra); // own/friend/sam/enemy
  if (in_array($rel,['own','friend']))   $s -= 1;
  if ($rel=='enemy')                     $s += 1;
  if (transitOnNatalSuryaMangalRashiOr7th($chart,$shaniTransit)) $s += 1;  // S10
  if (runningDashaLordBenefic($chart))   $s -= 1;  // S13
  $occ = sadesatiOccurrenceNumber($chart); // 1st/2nd/3rd via age
  if ($occ==2) $s -= 1;
  if ($occ>=3) $s += 1;                            // S11: तीसरी प्रायः घातक
  return max(0,min(3,$s));
}
```

| Mangal | 12 | — | वृषभ लग्न → धन वृद्धि; व्यापार से धन; स्त्री-पक्ष से धन। धनु लग्न → भाग्य वृद्धि; पुत्र उन्नति। | (निर्बल+पाप-पीड़ित) वृषभ लग्न → व्यापार में हानि, स्त्री-पक्ष से हानि। धनु लग्न → पुत्र के कारण बहुत व्यय; पुत्र के काम-धन्धे डांवाडोल। |  |
| Budh | 1 | मिथुन/कन्या | (अप्रभावित बुध) मान तथा धन; शास्त्रों का अध्ययन; तीव्र बुद्धि। नवमेश होकर लग्न में → धर्म-परोपकार में रत, धन-मान, मानसिक शान्ति, गाने-बजाने में रुचि। कन्या लग्न → अधिक मान (विशेषतः सूर्य-युक्त बलवान + शुभ प्रभाव)। | (लग्नेश निर्बल, पाप-दृष्ट) बुद्धिहीन; मस्तिष्क शोथ (Meningitis), पागलपन (Mania); मान-धन हानि; चर्म रोग; मिथुन लग्न → हाथों/सांस की नली में कष्ट; कन्या लग्न → अंतड़ियों में कष्ट। |  |
| Budh | 2 | वृषभ | वाक्-चतुराई तथा व्यापार से खूब धन; पुत्र से आर्थिक लाभ; लेखन से आमदनी; विद्या-प्राप्ति का अवसर। सिंह लग्न → आय विशेष बढ़े; बैंक आदि में नौकरी या सूदखोरी से पर्याप्त धन; बड़े भाई सहायक। | (निर्बल+पीड़ित) धन-हानि; पुत्र द्वारा भी धन-नाश; विद्या में असफल; जिह्वा-दोष — हकलाना या अपशब्द; लेखन-व्यापार में हानि (गलत टिप्पणी/गलत कागज़ पर हस्ताक्षर से नुकसान); सिंह लग्न → बड़े भाई-बहनों या दामाद से धन-हानि; ब्याज/निवेश आय कम। |  |
| Budh | 3 | मेष | धन की कमी; मित्रों से कुछ सहायता; परिश्रम बहुत। कर्क लग्न → व्यय अधिक; भाइयों-मित्रों पर धन व्यय। | (निर्बल) हाथ-कन्धे में चोट की सम्भावना; भाई-बन्धुओं से वैमनस्य; शत्रुओं से दबकर रहना; अकस्मात् शारीरिक कष्ट से जान पर; कर्क लग्न → भाइयों पर अधिक व्यय, मित्रों से हानि, यात्राएँ असफल — व्यर्थ समय व धन व्यय। |  |
| Budh | 4 | मीन/मिथुन | सुख; भूमि तथा वाहन। मिथुन लग्न → वही फल + विद्या-शास्त्राध्ययन में मन; व्यापार व लेखन से धन। | (निर्बल+पीड़ित) दुःख; मानसिक रोगों का शिकार; अकस्मात् शारीरिक कष्ट; सम्बन्धियों से हानि। मिथुन लग्न के लिए भी यही। | बुध पर मंगल तथा शुक्र का प्रभाव → माता से स्नेह अधिक; मन आमोद-प्रमोद में। |
| Budh | 5 | कुम्भ/वृषभ | सट्टे आदि से धन; भाग्य वृद्धि; प्रतिभा का अनुसन्धान में उपयोग। वृषभ लग्न → विद्या व भाग्य वृद्धि; पुत्रों से धन-सम्मान। | (निर्बल) पुत्रों से कष्ट; अनुसन्धान विफल; बुद्धि व धन का ह्रास; वृषभ लग्न → पुत्रों द्वारा धन-नाश; बुद्धि बिगड़ने से धन-नाश; विद्या-प्राप्ति फिर भी रहती है। |  |

| Budh | 8 | कुम्भ/वृश्चिक | कुम्भ लग्न → पैरा 5 जैसा फल। वृश्चिक लग्न → जातक बहुत धनी; दूर देश के व्यापार से धन; बड़े भाइयों से लाभ; यज्ञीय कार्यों में प्रवृत्ति। | (निर्बल+पीड़ित) धन-आय में कमी; बड़े बहिन-भाइयों से धन-हानि; अचानक विदेश यात्रा का संयोग; शरीर में कष्ट। |  |
| Budh | 9 | मकर | भाग्य वृद्धि; राज्य से सहायता; पिता के धन-मान में वृद्धि; धार्मिक कार्यों में मन; मामा से धन-लाभ। | (निर्बल+पीड़ित) शत्रुओं द्वारा चिरस्थायी हानि; पिता को मानहानि; मामा से आर्थिक हानि की संभावना; दूर स्थानों की यात्रा में बहुत कष्ट। |  |
| Budh | 10 | धनु | सम्मान और अधिकार; व्यापार में नाम; राज्य की कृपा। | (निर्बल+पीड़ित) राज्य-दण्ड स्वरूप नौकरी जाना; व्यापार से हानि; स्त्री के कारण अपयश की संभावना। |  |
| Budh | 11 | वृश्चिक | पैरा 8 जैसा फल (बलवान या निर्बल जैसा भी हो)। | वही cross-ref। |  |
| Budh | 12 | तुला | लाभ रहता है; धार्मिक कार्यों में व्यय; दूर की यात्राओं से लाभ। | (निर्बल+पीड़ित) हीनता; दूर स्थानों की यात्राओं पर व्यर्थ व्यय। |  |
| Guru | 1 | धनु | विशेष सुख; मन स्थिर व शान्त; माता, वाहन, जमीन-जायदाद का सुख; सर्वसाधारण से लाभ; धन-मान वृद्धि। | (निर्बल+पीड़ित) बहुत दुःख — माता, वाहन, जमीन-जायदाद सभी कारणों से; मानसिक रोगों की संभावना; सम्बन्धियों से कलह; सर्वसाधारण से विरोध। |  |
| Guru | 2 | वृश्चिक | आशातीत धन-लाभ; पुत्रों के भाग्य से भी धन; बुद्धि धन कमाने में सफल; योजनाएँ सुफल; विद्या वृद्धि; विवाह का सुख; धन शुभ कार्यों में व्यय। | (निर्बल+पीड़ित) बहुत धन-हानि; पुत्र द्वारा भी धन-नाश; सट्टे में नाश; विद्या में हीनता; कुटुम्ब से कलह; बौद्धिक गलतियों से बहुत नुकसान। |  |
| Guru | 3 | तुला | मित्रों और शत्रुओं दोनों से आंशिक लाभ; आर्थिक स्तर में विशेष अन्तर नहीं; धन-मान में सामान्य वृद्धि; मनोकामना विलम्ब से फलीभूत; शुभ कार्यों में प्रवृत्ति। | (निर्बल+पीड़ित) मित्रों, मामा और शत्रुओं से नुकसान; मित्रों तथा भाइयों से वैमनस्य। | गुरु और शुक्र इकट्ठे + पीड़ित → पेट और जिगर के रोगों की सम्भावना। |
| Guru | 4 | कन्या | असाधारण सुख; वाहन-सुख (विशेषतया गुरु के साथ शुक्र हो); व्यापार वृद्धि; स्त्री-कुण्डली → पति को विशेष मान-सम्मान; सम्बन्धी विशेष सहायक। | (निर्बल+पीड़ित) सुख का नाश; घर से दूर रहना; सम्बन्धियों से कलह; सामान्य-जन से शत्रुता; जमीन-जायदाद का लाभ नहीं। |  |
| Guru | 5 | सिंह | भाग्य वृद्धि; पुत्र को धन-सुख; योजनाओं में सफलता; अनुसन्धान में विशेष सफलता; धार्मिक रुचि। | (निर्बल+पीड़ित) धन-हानि; पुत्र के धन-सुख का नाश; विदेश यात्रा की सम्भावना; पुत्रों से विशेष परेशानी; योजनाएँ असफल। (पुस्तक-misprint C24 देखें) |  |
| Guru | 6 | कर्क | बलशाली शत्रुओं व विरोधियों का सामना; आय साधारण। | (निर्बल) शत्रुओं द्वारा जीवन तहस-नहस; धन-हानि बहुत; राज्य-कर्मचारी विरुद्ध; पिता को धन-मान का सुख कम; जातक रोगी। |  |
| Guru | 7 | मिथुन | स्त्री-कुण्डली → इससे बढ़कर अच्छी स्थिति नहीं — दीर्घजीवी, चरित्रवान, धनी पति। पुरुष → राज्य और व्यापार दोनों से लाभ। | (निर्बल+पीड़ित) राज्य से राजदण्ड व कष्ट; नौकरी तक छोड़नी पड़े; मान-हानि व धन-हानि; शारीरिक कष्ट विशेष। |  |
| Guru | 8 | वृषभ | धन की विशेष आय नहीं, पर बहुत कमी भी नहीं; बड़े भाई से कुछ लाभ; स्वास्थ्य ठीक। | (निर्बल+पीड़ित) धन कमाता है पर बड़े भाई से विरोध; काला धन पकड़े जाने से कर (Tax) की बहुत कठिनाई; विदेश यात्रा की संभावना। |  |
| Guru | 9 | मेष | भाग्य खूब चमकता है; हर प्रकार से राज्य-कृपा व सहायता; गुरुजनों की कृपा; पिता के धन के लिए भाग्यप्रद। | (निर्बल+पीड़ित) विशेष भाग्य-हानि; पिता को दुःख; राज्य से विरोध व दण्ड। | गुरु पर राहु तथा शनि का प्रभाव → नौकरी-व्यवसाय से हाथ धोना; भयंकर आर्थिक दुःख। |
| Guru | 10 | मीन | विशेष मान-सम्मान; राज्य से विशेष अनुकूलता व कृपा; शुभ व यज्ञीय कार्यों में प्रवृत्ति व विशेष लाभ। | (निर्बल+पाप-पीड़ित) राज्य-दण्ड व विरोध से दुःख; कार्यों में असफलता; धन और मान दोनों का नाश; कार्यों में अवैधता व अधार्मिकता का समावेश। |  |
| Guru | 11 | कुम्भ | विशेष रूप से व अनायास धन-लाभ; ब्याज, बैंक, कैश सर्टिफिकेट्स से विशेष आय; बड़े भाई से आय। | (निर्बल+पीड़ित) आय में बहुत कमी; जातक रोगी; बड़े भाई द्वारा आर्थिक हानि; स्त्री को भी कष्ट। |  |
| Guru | 12 | मकर | विशेष धन-लाभ नहीं; बहिन-भाइयों पर उनकी भलाई के लिए व्यय; राज्य से कुछ परेशानी (टैक्स आदि)। | (निर्बल+पीड़ित) व्यर्थ व्यय अधिक; भाइयों से कष्ट; परन्तु आय अच्छी रहती है। |  |
| Shukra | 1 | तुला | धन तथा मान; विलास के साधन खूब; स्त्रियों से सम्पर्क; विलास-वस्तुओं के व्यापार से लाभ। | (निर्बल+पाप-पीड़ित) भारी शारीरिक कष्ट — मृत्यु-तुल्य; विदेश यात्रा की संभावना; पत्नी से वैमनस्य; स्त्री वर्ग से व विलास-वस्तुओं के व्यापार में हानि। |  |
| Shukra | 2 | कन्या | खूब धन-लाभ; भाग्य बढ़ता है; परन्तु जातक रोगी भी रहता है; कुटुम्बी जनों की सहायता से पैरों पर खड़ा होना। | (निर्बल+पाप-पीड़ित) धन-नाश; भाग्य में धक्का; पत्नी को कष्ट; कुटुम्ब से अनबन; विद्या-बुद्धि में असफलता। |  |
| Shukra | 3 | सिंह | मित्रों से सहायता; छोटे भाइयों की उन्नति; पास-पड़ौस से सुख; पराक्रम का परिचय; धन-प्राप्ति साधारण। | (निर्बल+पाप-पीड़ित) मित्रों-भाइयों से कोई सहायता नहीं; परन्तु धन की आय बहुत अच्छी; मित्रों के कारण कार्यों में सफलता। |  |
| Shukra | 4 | कर्क | वाहन-सुख; सुन्दर मकान निवास हेतु; लाड़-प्यार में वृद्धि; सम्बन्धियों से सहयोग; माता से सुख। | (निर्बल+पाप-पीड़ित) किसी सम्बन्धी से सहायता/सुख नहीं; घर में टिकना नहीं; नौकरीपेशा के तबादले बहुत; माता से सुख कम। |  |
| Shukra | 5 | मिथुन | धन-लाभ अधिक; पुत्र की ओर से सुख व सहायता; कुछ जातकों के अन्य स्त्रियों से प्रेम-सम्बन्ध भी। | (निर्बल+पाप-पीड़ित) धन-नाश; बुद्धि मन्द; पुत्र से कष्ट; सट्टे आदि के व्यापार में घाटा। |  |
| Shukra | 6 | वृषभ | धन की बहुतायत; मित्रों तथा शत्रुओं दोनों से धन; परिश्रम से मान-धन वृद्धि; मामा से धन-लाभ। | (निर्बल+पाप-पीड़ित) एक ओर पदोन्नति का लाभ, दूसरी ओर रोग भी काफी; शत्रुओं के गुप्त वार से पदोन्नति में बाधा। |  |
| Shukra | 7 | मेष | विवाह-सुख या स्त्री-भोग का योग; व्यापार-वृद्धि, धन बढ़ता; स्त्री वर्ग तथा ससुराल से लाभ; पुरुषार्थ वृद्धि। | (निर्बल+पाप-पीड़ित) पत्नी बहुत रोगिणी; धन-नाश; व्यापार में हानि; राज्य से परेशानी। |  |
